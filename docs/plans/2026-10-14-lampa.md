# lampa Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Go TUI that listens to macOS system (or one app's) audio and drives a Yeelight W3 (music mode) and WiZ bulbs in real time, with live spectrum/beat view and on-the-fly mode, palette, sensitivity and latency controls.

**Architecture:** A tiny Swift helper (`lampa-tap`) captures audio via Core Audio Process Tap and streams a sample-rate header + mono float32 PCM to stdout. The Go binary reads it (`audio`), extracts `Features` (`analysis`), turns them into per-lamp colors (`modes` + `color`), smooths and ships them to lamps through per-lamp latest-frame runners (`light`, `wiz`, `yeelight`), and renders everything in Bubble Tea (`tui`). `engine` glues these; `cmd/lampa` wires config, discovery and CLI.

**Tech Stack:** Go 1.27, `github.com/charmbracelet/bubbletea@v1.3.10`, `github.com/charmbracelet/lipgloss@v1.1.0`, `gonum.org/v1/gonum@v0.17.0` (FFT), `github.com/BurntSushi/toml@v1.6.0`; Swift 6.4 toolchain with `swift-tools-version:5.10` (language mode 5), CoreAudio.

**Spec:** `docs/specs/2026-10-14-lampa-design.md`

## Global Constraints

- Module path: `lampa`. All Go packages under `internal/`, binary in `cmd/lampa`.
- macOS ≥ 14.4 (Core Audio Process Tap). Developed on macOS 27, arm64.
- Tap stream protocol: first 4 bytes = sample rate as little-endian uint32, then little-endian float32 mono samples. Logs only on stderr.
- Analysis: Hann window 2048, hop 512; 24 log-spaced bands 30 Hz–16 kHz (deviation from spec's "mel": mel spacing leaves no bands under 60 Hz; spec updated in Task 2).
- Groups: sub < 60 Hz, bass 60–250, lowmid 250–1k, mid 1k–4k, high ≥ 4k (by band centre).
- Normalizer: instant attack, decay τ = 6 s, 30 dB range, absolute floor −80 dB.
- Onset threshold = median(last 1.5 s) × k + min, k = 1 + (1 − sensitivity); refractory 80 ms.
- Tempo: 70–180 BPM, prior centred on 120; `Confidence < 0.4` ⇒ modes use onsets instead of beat grid.
- Latency default 60 ms, adjustable −200…+200 ms in 10 ms steps.
- WiZ: UDP 38899, `setPilot {r,g,b,dimming}`, dimming 10–100, default 20 fps, heartbeat `getPilot` every 1 s, lost after 3 s silence.
- Yeelight: control TCP 55443, `set_music [1, ip, port]`, then `set_rgb`/`set_bright` on the bulb's callback connection, default 20 fps; reconnect backoff 0.5 s → 8 s; without `set_music` support: 1 fps with `smooth` 900 ms.
- Colors interpolated in OKLCH; brightness gamma 2.0 applied in `light.Split`.
- Config: `~/.config/lampa/config.toml`; missing file is not an error.
- On exit (q, Ctrl-C, SIGTERM) every lamp that was snapshotted is restored; Yeelight leaves music mode.
- One lamp's failure never stops other lamps or the TUI.
- `go test ./... -race` passes at the end of every task.
- Commit author for this repo is already configured (Aleph). Every commit message ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```go
// Package color does perceptual color math (OKLCH) and smoothing for lamps.
package color

import (
	"fmt"
	"math"
)

// RGB is gamma-encoded sRGB, components 0..1.
type RGB struct{ R, G, B float64 }

// OKLCH: L 0..1, C >= 0 (≈0.37 max in sRGB), H in degrees [0,360).
type OKLCH struct{ L, C, H float64 }

func clamp01(x float64) float64 {
	if math.IsNaN(x) || x < 0 {
		return 0
	}
	if x > 1 {
		return 1
	}
	return x
}

func (c RGB) Bytes() (r, g, b uint8) {
	q := func(x float64) uint8 { return uint8(math.Round(clamp01(x) * 255)) }
	return q(c.R), q(c.G), q(c.B)
}

func (c RGB) Hex() string {
	r, g, b := c.Bytes()
	return fmt.Sprintf("#%02x%02x%02x", r, g, b)
}

func toLinear(x float64) float64 {
	if x <= 0.04045 {
		return x / 12.92
	}
	return math.Pow((x+0.055)/1.055, 2.4)
}

func toGamma(x float64) float64 {
	if x <= 0.0031308 {
		return 12.92 * x
	}
	return 1.055*math.Pow(x, 1/2.4) - 0.055
}

func FromRGB(c RGB) OKLCH {
	r, g, b := toLinear(c.R), toLinear(c.G), toLinear(c.B)
	l := math.Cbrt(0.4122214708*r + 0.5363325363*g + 0.0514459929*b)
	m := math.Cbrt(0.2119034982*r + 0.6806995451*g + 0.1073969566*b)
	s := math.Cbrt(0.0883024619*r + 0.2817188376*g + 0.6299787005*b)
	L := 0.2104542553*l + 0.7936177850*m - 0.0040720468*s
	A := 1.9779984951*l - 2.4285922050*m + 0.4505937099*s
	B := 0.0259040371*l + 0.7827717662*m - 0.8086757660*s
	h := math.Atan2(B, A) * 180 / math.Pi
	if h < 0 {
		h += 360
	}
	return OKLCH{L: L, C: math.Hypot(A, B), H: h}
}

// linear returns linear-light sRGB, possibly out of [0,1].
func (c OKLCH) linear() (r, g, b float64) {
	h := c.H * math.Pi / 180
	A, B := c.C*math.Cos(h), c.C*math.Sin(h)
	l := c.L + 0.3963377774*A + 0.2158037573*B
	m := c.L - 0.1055613458*A - 0.0638541728*B
	s := c.L - 0.0894841775*A - 1.2914855480*B
	l, m, s = l*l*l, m*m*m, s*s*s
	r = 4.0767416621*l - 3.3077115913*m + 0.2309699292*s
	g = -1.2684380046*l + 2.6097574011*m - 0.3413193965*s
	b = -0.0041960863*l - 0.7034186147*m + 1.7076147010*s
	return
}

func inGamut(r, g, b float64) bool {
	const eps = 1e-6
	return r >= -eps && r <= 1+eps && g >= -eps && g <= 1+eps && b >= -eps && b <= 1+eps
}

// ToRGB converts to sRGB, lowering chroma (keeping L and H) until in gamut.
func (c OKLCH) ToRGB() RGB {
	c.L = clamp01(c.L)
	if c.C < 0 || math.IsNaN(c.C) {
		c.C = 0
	}
	r, g, b := c.linear()
	if !inGamut(r, g, b) {
		lo, hi := 0.0, c.C
		for i := 0; i < 20; i++ {
			mid := (lo + hi) / 2
			if rr, gg, bb := (OKLCH{c.L, mid, c.H}).linear(); inGamut(rr, gg, bb) {
				lo = mid
			} else {
				hi = mid
			}
		}
		r, g, b = (OKLCH{c.L, lo, c.H}).linear()
	}
	return RGB{toGamma(clamp01(r)), toGamma(clamp01(g)), toGamma(clamp01(b))}
}

// Mix interpolates L, C linearly and H along the shortest arc.
func Mix(a, b OKLCH, t float64) OKLCH {
	dh := math.Mod(b.H-a.H+540, 360) - 180
	h := math.Mod(a.H+dh*t+360, 360)
	return OKLCH{L: a.L + (b.L-a.L)*t, C: a.C + (b.C-a.C)*t, H: h}
}
```

```go
package color

import "math"

// Palette is a cyclic gradient: At(1) == At(0).
type Palette struct {
	Name  string
	Stops []OKLCH
}

func (p Palette) At(t float64) OKLCH {
	n := len(p.Stops)
	if n == 0 {
		return OKLCH{L: 1}
	}
	t = t - math.Floor(t)
	x := t * float64(n)
	i := int(x) % n
	return Mix(p.Stops[i], p.Stops[(i+1)%n], x-math.Floor(x))
}

var Palettes = []Palette{
	{"sunset", []OKLCH{{0.70, 0.20, 30}, {0.75, 0.17, 60}, {0.62, 0.22, 350}, {0.55, 0.20, 310}}},
```

```go
package color

import (
	"math"
	"time"
)

// Envelope follows a target with separate rise (Attack) and fall (Release)
// time constants; zero means instant.
type Envelope struct {
	Attack, Release time.Duration
	v               float64
}

func (e *Envelope) Value() float64 { return e.v }

func (e *Envelope) Step(target float64, dt time.Duration) float64 {
	tau := e.Release
	if target > e.v {
		tau = e.Attack
	}
	if tau <= 0 {
		e.v = target
	} else {
		e.v += (target - e.v) * (1 - math.Exp(-float64(dt)/float64(tau)))
	}
	return e.v
}
```

```go
package analysis

import "math"

func sine(freq, amp, seconds float64, rate int) []float32 {
	n := int(seconds * float64(rate))
	out := make([]float32, n)
	for i := range out {
		out[i] = float32(amp * math.Sin(2*math.Pi*freq*float64(i)/float64(rate)))
	}
	return out
}

func silence(seconds float64, rate int) []float32 {
	return make([]float32, int(seconds*float64(rate)))
}

// burst adds a decaying sine burst of length dur starting at sample start.
func burst(dst []float32, start int, freq, amp, dur float64, rate int) {
	n := int(dur * float64(rate))
	for i := 0; i < n && start+i < len(dst); i++ {
		t := float64(i) / float64(rate)
		env := math.Exp(-t / (dur / 4))
		dst[start+i] += float32(amp * env * math.Sin(2*math.Pi*freq*t))
	}
}

// metronome: kick (60 Hz, 120 ms) on every beat for the given duration.
// Returns the samples and the true beat times in seconds.
func metronome(bpm, seconds float64, rate int) ([]float32, []float64) {
	out := silence(seconds, rate)
	var beats []float64
	period := 60 / bpm
	for t := 0.5; t < seconds; t += period {
		burst(out, int(t*float64(rate)), 60, 0.8, 0.12, rate)
		burst(out, int(t*float64(rate)), 3000, 0.2, 0.02, rate)
		beats = append(beats, t)
	}
	return out, beats
}

func pushAll(a *Analyzer, s []float32, chunk int) []Features {
	var fs []Features
	for i := 0; i < len(s); i += chunk {
		j := min(i+chunk, len(s))
		fs = append(fs, a.Push(s[i:j])...)
	}
	return fs
}
```

```go
package analysis

import (
	"math"
	"math/cmplx"

	"gonum.org/v1/gonum/dsp/fourier"
)

// Spectrum computes a Hann-windowed magnitude spectrum and log-spaced band powers.
type Spectrum struct {
	rate     int
	fft      *fourier.FFT
	window   []float64
	buf      []float64
	coeffs   []complex128
	Mag      []float64 // WindowSize/2+1 bins; full-scale sine peak ≈ 1
	bandBins [NumBands][2]int
	Centers  [NumBands]float64
}

func NewSpectrum(rate int) *Spectrum {
	s := &Spectrum{
		rate:   rate,
		fft:    fourier.NewFFT(WindowSize),
		window: make([]float64, WindowSize),
		buf:    make([]float64, WindowSize),
		Mag:    make([]float64, WindowSize/2+1),
	}
	for i := range s.window {
		s.window[i] = 0.5 - 0.5*math.Cos(2*math.Pi*float64(i)/float64(WindowSize))
	}
	ratio := MaxFreq / MinFreq
	for b := 0; b < NumBands; b++ {
		lo := MinFreq * math.Pow(ratio, float64(b)/NumBands)
		hi := MinFreq * math.Pow(ratio, float64(b+1)/NumBands)
		s.Centers[b] = math.Sqrt(lo * hi)
		kLo := int(math.Ceil(lo / s.binWidth()))
		kHi := int(math.Ceil(hi / s.binWidth()))
		if kHi > len(s.Mag) {
			kHi = len(s.Mag)
		}
		if kHi <= kLo { // band narrower than a bin: use the nearest bin
			k := int(math.Round(s.Centers[b] / s.binWidth()))
			kLo, kHi = k, k+1
		}
		s.bandBins[b] = [2]int{kLo, kHi}
	}
	return s
}

func (s *Spectrum) binWidth() float64      { return float64(s.rate) / WindowSize }
func (s *Spectrum) BinFreq(k int) float64 { return float64(k) * s.binWidth() }

// Compute analyses exactly WindowSize samples.
func (s *Spectrum) Compute(frame []float64) {
	for i, x := range frame {
		s.buf[i] = x * s.window[i]
	}
	s.coeffs = s.fft.Coefficients(s.coeffs, s.buf)
	for k, c := range s.coeffs {
		s.Mag[k] = cmplx.Abs(c) / (WindowSize / 4)
	}
}

// Bands writes the power (sum of squared magnitudes) of each band.
func (s *Spectrum) Bands(out *[NumBands]float64) {
	for b, r := range s.bandBins {
		p := 0.0
		for k := r[0]; k < r[1]; k++ {
			p += s.Mag[k] * s.Mag[k]
		}
		out[b] = p
	}
}
```

```go
package analysis

import (
	"math"
	"time"
)

const (
	floorDB  = -80.0
	rangeDB  = 30.0
	decayTau = 6 * time.Second
)

// Normalizer maps a power to 0..1 relative to a slowly decaying peak,
// so quiet and loud tracks both use the full range.
type Normalizer struct {
	peak float64 // linear power
}

func toDB(p float64) float64 { return 10 * math.Log10(p+1e-20) }

// PeakDB is the current ceiling in dB (never below floorDB+rangeDB).
func (n *Normalizer) PeakDB() float64 { return math.Max(toDB(n.peak), floorDB+rangeDB) }

// Step updates the peak and returns the normalized value. minPeakDB lets the
// caller raise the ceiling (e.g. relative to the loudest band) so leakage in
// an otherwise empty band does not get amplified to full scale.
func (n *Normalizer) Step(power float64, dt time.Duration, minPeakDB float64) float64 {
	n.peak = math.Max(power, n.peak*math.Exp(-float64(dt)/float64(decayTau)))
	x := toDB(power)
	if x < floorDB {
		return 0
	}
	peak := math.Max(n.PeakDB(), minPeakDB)
	return clamp01(1 - (peak-x)/rangeDB)
}

func clamp01(x float64) float64 {
	if math.IsNaN(x) || x < 0 {
		return 0
	}
	if x > 1 {
		return 1
	}
	return x
}
```

```go
package analysis

import (
	"math"
	"time"
)

// A band may not be boosted more than this far below the loudest band's peak.
const bandHeadroomDB = 24.0

type Config struct {
	SampleRate  int
	Sensitivity float64       // 0..1, default 0.5
	Latency     time.Duration // beats are predicted this far ahead
}

type Analyzer struct {
	cfg     Config
	spec    *Spectrum
	buf     []float64
	pending int
	total   int64 // samples consumed
	dt      time.Duration
	frame   []float64

	power    [NumBands]float64
	bandNorm [NumBands]Normalizer
	loudNorm Normalizer
	groupOf  [NumBands]int // 0 sub, 1 bass, 2 lowmid, 3 mid, 4 high
	groupN   [5]int
}

func New(cfg Config) *Analyzer {
	if cfg.SampleRate <= 0 {
		cfg.SampleRate = 48000
	}
	if cfg.Sensitivity == 0 {
		cfg.Sensitivity = 0.5
	}
	a := &Analyzer{
		cfg:   cfg,
		spec:  NewSpectrum(cfg.SampleRate),
		dt:    time.Duration(float64(HopSize) / float64(cfg.SampleRate) * float64(time.Second)),
		frame: make([]float64, WindowSize),
	}
	for b, c := range a.spec.Centers {
		g := 4
		switch {
		case c < 60:
			g = 0
		case c < 250:
			g = 1
		case c < 1000:
			g = 2
		case c < 4000:
			g = 3
		}
		a.groupOf[b] = g
		a.groupN[g]++
	}
	return a
}

func (a *Analyzer) Rate() int                       { return a.cfg.SampleRate }
func (a *Analyzer) SetSensitivity(s float64)        { a.cfg.Sensitivity = clamp01(s) }
func (a *Analyzer) SetLatency(d time.Duration)      { a.cfg.Latency = d }

func sanitize(x float32) float64 {
	v := float64(x)
	if math.IsNaN(v) || math.IsInf(v, 0) {
		return 0
	}
	return math.Max(-1, math.Min(1, v))
}

// Push consumes samples and returns one Features per completed hop.
func (a *Analyzer) Push(samples []float32) []Features {
	var out []Features
	for _, x := range samples {
		a.buf = append(a.buf, sanitize(x))
		a.total++
		a.pending++
		if len(a.buf) >= WindowSize && a.pending >= HopSize {
			copy(a.frame, a.buf[len(a.buf)-WindowSize:])
			out = append(out, a.process())
			a.pending = 0
			if len(a.buf) > 4*WindowSize {
				a.buf = append(a.buf[:0], a.buf[len(a.buf)-WindowSize:]...)
			}
		}
	}
	return out
}

func (a *Analyzer) now() time.Duration {
	center := float64(a.total) - WindowSize/2
	return time.Duration(center / float64(a.cfg.SampleRate) * float64(time.Second))
}

func (a *Analyzer) process() Features {
	f := Features{Time: a.now()}
	a.spec.Compute(a.frame)
	a.computeBands(&f)
	return f
}

func (a *Analyzer) computeBands(f *Features) {
	a.spec.Bands(&a.power)
	loudest := floorDB
	total := 0.0
	for b := range a.bandNorm {
		loudest = math.Max(loudest, a.bandNorm[b].PeakDB())
		total += a.power[b]
	}
	var sums [5]float64
	for b := range a.bandNorm {
		v := a.bandNorm[b].Step(a.power[b], a.dt, loudest-bandHeadroomDB)
		f.Bands[b] = v
		sums[a.groupOf[b]] += v
	}
	avg := func(g int) float64 {
		if a.groupN[g] == 0 {
			return 0
		}
		return sums[g] / float64(a.groupN[g])
	}
	f.Groups = Groups{Sub: avg(0), Bass: avg(1), LowMid: avg(2), Mid: avg(3), High: avg(4)}
	f.Loudness = a.loudNorm.Step(total, a.dt, floorDB+rangeDB)
}
```

```go
package analysis

import (
	"math"
	"slices"
	"time"
)

const (
	onsetGamma      = 20.0                   // log compression: log1p(gamma*mag)
	onsetMinFlux    = 1.0                    // absolute floor of the threshold
	onsetHistory    = 1500 * time.Millisecond // median window
	onsetRefractory = 80 * time.Millisecond
	lowCutHz        = 250.0
	highCutHz       = 1000.0
)

type fluxBand struct {
	lo, hi  int // bin range [lo,hi)
	hist    []float64
	pos     int
	filled  bool
	lastHit time.Duration
	scratch []float64
}

func (b *fluxBand) median() float64 {
	n := len(b.hist)
	if !b.filled {
		n = b.pos
	}
	if n == 0 {
		return 0
	}
	b.scratch = append(b.scratch[:0], b.hist[:n]...)
	slices.Sort(b.scratch)
	return b.scratch[n/2]
}

func (b *fluxBand) push(v float64) {
	b.hist[b.pos] = v
	b.pos++
	if b.pos == len(b.hist) {
		b.pos, b.filled = 0, true
	}
}

type onsetDetector struct {
	prev      []float64
	havePrev  bool
	low, high fluxBand
}

func newOnsetDetector(spec *Spectrum, rate int) *onsetDetector {
	hist := int(onsetHistory.Seconds() * float64(rate) / HopSize)
	bins := len(spec.Mag)
	lowHi := int(math.Ceil(lowCutHz / spec.binWidth()))
	highLo := int(math.Ceil(highCutHz / spec.binWidth()))
	return &onsetDetector{
		prev: make([]float64, bins),
		low:  fluxBand{lo: 1, hi: lowHi, hist: make([]float64, hist), lastHit: -time.Hour},
		high: fluxBand{lo: highLo, hi: bins, hist: make([]float64, hist), lastHit: -time.Hour},
	}
}

func (d *onsetDetector) step(mag []float64, now time.Duration, k float64) (low, high Onset, flux float64) {
	fluxOf := func(b *fluxBand) float64 {
		s := 0.0
		for i := b.lo; i < b.hi; i++ {
			if diff := math.Log1p(onsetGamma*mag[i]) - d.prev[i]; diff > 0 {
				s += diff
			}
		}
		return s
	}
	var fl, fh float64
	if d.havePrev {
		fl, fh = fluxOf(&d.low), fluxOf(&d.high)
	}
	for i, m := range mag {
		d.prev[i] = math.Log1p(onsetGamma * m)
	}
	d.havePrev = true
	return d.decide(&d.low, fl, now, k), d.decide(&d.high, fh, now, k), fl + fh
}

func (d *onsetDetector) decide(b *fluxBand, flux float64, now time.Duration, k float64) Onset {
	thr := b.median()*k + onsetMinFlux
	b.push(flux)
	if flux <= thr || now-b.lastHit < onsetRefractory {
		return Onset{}
	}
	b.lastHit = now
	return Onset{Hit: true, Strength: clamp01((flux - thr) / thr)}
}
```

```go
package analysis

import (
	"math"
	"time"
)

const (
	tempoWindow   = 6 * time.Second
	tempoMinBPM   = 70.0
	tempoMaxBPM   = 180.0
	tempoPriorBPM = 120.0
	tempoPriorOct = 0.9 // width of the log2 prior, octaves
	pllGain       = 0.25
	// onsetBias corrects the systematic offset between when the detector
	// fires and the true onset (frame time is the window centre).
	onsetBias = 0 * time.Millisecond
)

type tempoTracker struct {
	fps     float64
	env     []float64
	pos     int
	filled  bool
	frames  int
	period  time.Duration
	conf    float64
	next    time.Duration
	beats   int
	lastOn  time.Duration
}

func newTempoTracker(fps float64) *tempoTracker {
	return &tempoTracker{fps: fps, env: make([]float64, int(tempoWindow.Seconds()*fps))}
}

func (t *tempoTracker) ordered() []float64 {
	if !t.filled {
		return t.env[:t.pos]
	}
	return append(append([]float64{}, t.env[t.pos:]...), t.env[:t.pos]...)
}

// estimate updates period and confidence from the onset envelope.
func (t *tempoTracker) estimate() {
	x := t.ordered()
	if len(x) < int(4*t.fps) {
		return
	}
	mean := 0.0
	for _, v := range x {
		mean += v
	}
	mean /= float64(len(x))
	y := make([]float64, len(x))
	acf0 := 0.0
	for i, v := range x {
		y[i] = v - mean
		acf0 += y[i] * y[i]
	}
	if acf0 < 1e-9 {
		t.period, t.conf = 0, 0
		return
	}
	lagMin := int(math.Floor(60 * t.fps / tempoMaxBPM))
	lagMax := int(math.Ceil(60 * t.fps / tempoMinBPM))
	r := make([]float64, lagMax+2)
	for lag := lagMin - 1; lag <= lagMax+1; lag++ {
		s := 0.0
		for i := 0; i+lag < len(y); i++ {
			s += y[i] * y[i+lag]
		}
		r[lag] = s / acf0 * float64(len(y)) / float64(len(y)-lag)
	}
	best, bestScore, sum := 0, math.Inf(-1), 0.0
	for lag := lagMin; lag <= lagMax; lag++ {
		bpm := 60 * t.fps / float64(lag)
		w := math.Exp(-0.5 * math.Pow(math.Log2(bpm/tempoPriorBPM)/tempoPriorOct, 2))
		if s := r[lag] * w; s > bestScore {
			best, bestScore = lag, s
		}
		sum += r[lag]
	}
	avg := sum / float64(lagMax-lagMin+1)
	// parabolic refinement
	lag := float64(best)
	if a, b, c := r[best-1], r[best], r[best+1]; a-2*b+c < 0 {
		lag += 0.5 * (a - c) / (a - 2*b + c)
	}
	t.conf = clamp01((r[best] - avg) * 2)
	p := time.Duration(lag / t.fps * float64(time.Second))
	if t.period == 0 || math.Abs(float64(p-t.period))/float64(t.period) > 0.05 {
		t.period = p
		t.next = t.lastOn + p
	} else {
		t.period = (t.period*3 + p) / 4
	}
}

func (t *tempoTracker) step(now time.Duration, flux float64, onset bool, latency time.Duration) (bpm, phase, bar, conf float64, beat bool) {
	t.env[t.pos] = flux
	t.pos++
	if t.pos == len(t.env) {
		t.pos, t.filled = 0, true
	}
	t.frames++
	if t.frames%int(t.fps/2) == 0 {
		t.estimate()
	}
	if onset {
		t.lastOn = now - onsetBias
	}
	if t.period == 0 || t.conf == 0 {
		return 0, 0, 0, t.conf, false
	}
	if onset {
		on := now - onsetBias
		e1, e2 := on-(t.next-t.period), on-t.next
		e := e1
		if absDur(e2) < absDur(e1) {
			e = e2
		}
		if absDur(e) < t.period/4 {
			t.next += time.Duration(pllGain * float64(e))
		}
	}
	ahead := now + latency
	for ahead >= t.next {
		beat = true
		t.beats++
		t.next += t.period
	}
	phase = clamp01(1 - float64(t.next-ahead)/float64(t.period))
	bar = (float64(t.beats%4) + phase) / 4
	return 60 / t.period.Seconds(), phase, bar, t.conf, beat
}

func absDur(d time.Duration) time.Duration {
	if d < 0 {
		return -d
	}
	return d
}

// buildMeter: how much louder the last ~1 s is than the last ~8 s (6 dB → 1).
type buildMeter struct {
	fast, slow float64
	init       bool
}

func (b *buildMeter) step(powerDB float64, dt time.Duration) float64 {
	powerDB = math.Max(powerDB, floorDB)
	if !b.init {
		b.fast, b.slow, b.init = powerDB, powerDB, true
	}
	b.fast += (powerDB - b.fast) * (1 - math.Exp(-dt.Seconds()/1.0))
	b.slow += (powerDB - b.slow) * (1 - math.Exp(-dt.Seconds()/8.0))
	return clamp01((b.fast - b.slow) / 6)
}
```

```go
package audio

import (
```

```go
func ReadWAV(r io.Reader) (int, []float32, error) {
	var hdr [12]byte
	if _, err := io.ReadFull(r, hdr[:]); err != nil {
		return 0, nil, fmt.Errorf("wav header: %w", err)
	}
	if string(hdr[0:4]) != "RIFF" || string(hdr[8:12]) != "WAVE" {
		return 0, nil, errors.New("not a RIFF/WAVE file")
	}
	var format, channels, bits uint16
	var rate uint32
	for {
		var ch [8]byte
		if _, err := io.ReadFull(r, ch[:]); err != nil {
			return 0, nil, fmt.Errorf("wav: no data chunk: %w", err)
		}
		id, size := string(ch[0:4]), binary.LittleEndian.Uint32(ch[4:8])
		switch id {
		case "fmt ":
			buf := make([]byte, size)
			if _, err := io.ReadFull(r, buf); err != nil {
				return 0, nil, err
			}
			format = binary.LittleEndian.Uint16(buf[0:2])
			channels = binary.LittleEndian.Uint16(buf[2:4])
			rate = binary.LittleEndian.Uint32(buf[4:8])
			bits = binary.LittleEndian.Uint16(buf[14:16])
		case "data":
			if channels == 0 {
				return 0, nil, errors.New("wav: data before fmt")
			}
			buf := make([]byte, size)
			n, err := io.ReadFull(r, buf)
			if err != nil && !errors.Is(err, io.ErrUnexpectedEOF) {
				return 0, nil, err
			}
			s, err := decodePCM(buf[:n], format, channels, bits)
			return int(rate), s, err
		default:
			if _, err := io.CopyN(io.Discard, r, int64(size+size%2)); err != nil {
				return 0, nil, err
			}
		}
	}
}

func decodePCM(b []byte, format, channels, bits uint16) ([]float32, error) {
	var width int
	var read func([]byte) float32
	switch {
	case format == 1 && bits == 16:
		width = 2
		read = func(p []byte) float32 { return float32(int16(binary.LittleEndian.Uint16(p))) / 32768 }
	case format == 3 && bits == 32:
		width = 4
		read = func(p []byte) float32 { return math.Float32frombits(binary.LittleEndian.Uint32(p)) }
	default:
		return nil, fmt.Errorf("wav: unsupported format %d/%d bits", format, bits)
	}
	frame := width * int(channels)
	out := make([]float32, len(b)/frame)
	for i := range out {
		var sum float32
		for c := 0; c < int(channels); c++ {
			off := i*frame + c*width
			sum += read(b[off : off+width])
		}
		out[i] = sum / float32(channels)
	}
	return out, nil
}

type WAVSource struct {
	Rate     int
	Samples  []float32
	Realtime bool
}

func (w *WAVSource) Run(ctx context.Context, out chan<- Block) error {
	const n = 512
	tick := time.Duration(float64(n) / float64(w.Rate) * float64(time.Second))
	next := time.Now()
	for i := 0; i < len(w.Samples); i += n {
		if w.Realtime {
			next = next.Add(tick)
			select {
			case <-time.After(time.Until(next)):
			case <-ctx.Done():
				return nil
			}
		}
		j := min(i+n, len(w.Samples))
		if !send(ctx, out, Block{Rate: w.Rate, Samples: w.Samples[i:j]}) {
			return nil
		}
	}
	return nil
}
```

```go
// decodeStream reads the lampa-tap protocol and emits blocks of up to
// `block` samples. A trailing partial sample is dropped.
func decodeStream(r io.Reader, block int, emit func(Block)) error {
	br := bufio.NewReaderSize(r, 64*1024)
	var hdr [4]byte
	if _, err := io.ReadFull(br, hdr[:]); err != nil {
		return fmt.Errorf("tap header: %w", err)
	}
	rate := int(binary.LittleEndian.Uint32(hdr[:]))
	if rate < 8000 || rate > 384000 {
		return fmt.Errorf("tap: implausible sample rate %d", rate)
	}
	buf := make([]byte, block*4)
	for {
		n, err := io.ReadFull(br, buf)
		if k := n / 4; k > 0 {
			s := make([]float32, k)
			for i := range s {
				s[i] = math.Float32frombits(binary.LittleEndian.Uint32(buf[i*4:]))
			}
			emit(Block{Rate: rate, Samples: s})
		}
		if errors.Is(err, io.EOF) || errors.Is(err, io.ErrUnexpectedEOF) {
			return nil
		}
		if err != nil {
			return err
		}
	}
}

// TapSource runs the lampa-tap helper; PID 0 means the whole system.
type TapSource struct {
	Path    string
	PID     int
	OnError func(error)
}

func (t *TapSource) report(err error) {
	if err != nil && t.OnError != nil {
		t.OnError(err)
	}
}

func (t *TapSource) Run(ctx context.Context, out chan<- Block) error {
	for ctx.Err() == nil {
		t.report(t.once(ctx, out))
		select {
		case <-ctx.Done():
		case <-time.After(time.Second):
		}
	}
	return nil
}

func (t *TapSource) once(ctx context.Context, out chan<- Block) error {
	args := []string{"capture"}
	if t.PID != 0 {
		args = append(args, "--pid", strconv.Itoa(t.PID))
	}
	cmd := exec.CommandContext(ctx, t.Path, args...)
	var stderr bytes.Buffer
	cmd.Stderr = &stderr
	stdout, err := cmd.StdoutPipe()
	if err != nil {
		return err
	}
	if err := cmd.Start(); err != nil {
		return fmt.Errorf("start %s: %w", t.Path, err)
	}
	derr := decodeStream(stdout, 512, func(b Block) { send(ctx, out, b) })
	werr := cmd.Wait()
	if ctx.Err() != nil {
		return nil
	}
	if msg := lastLine(stderr.String()); msg != "" {
		return errors.New(msg) // e.g. the permission hint printed by lampa-tap
	}
	if derr != nil {
		return derr
	}
	if werr != nil {
		return fmt.Errorf("lampa-tap exited: %w", werr)
	}
	return errors.New("lampa-tap exited")
}

func lastLine(s string) string {
	lines := strings.Split(strings.TrimSpace(s), "\n")
	return strings.TrimSpace(lines[len(lines)-1])
}
```

```swift
        let outputUID = try getString(output, kAudioDevicePropertyDeviceUID)

            kAudioAggregateDeviceUIDKey: UUID().uuidString,
            kAudioAggregateDeviceMainSubDeviceKey: outputUID,
            kAudioAggregateDeviceIsPrivateKey: true,
            kAudioAggregateDeviceIsStackedKey: false,
            kAudioAggregateDeviceTapAutoStartKey: true,
            kAudioAggregateDeviceSubDeviceListKey: [[kAudioSubDeviceUIDKey: outputUID]],
            kAudioAggregateDeviceTapListKey: [[
                kAudioSubTapDriftCompensationKey: true,
                kAudioSubTapUIDKey: desc.uuid.uuidString,
            ]],
        ]
        try check(AudioHardwareCreateAggregateDevice(dict as CFDictionary, &aggID), "create aggregate device")

        var rate = UInt32(asbd.mSampleRate).littleEndian
        writeAll(Data(bytes: &rate, count: 4))

        try check(AudioDeviceCreateIOProcIDWithBlock(&procID, aggID, nil) { [weak self] _, input, _, _, _ in
            self?.handle(input, nonInterleaved: nonInterleaved)
        }, "create IO proc")
        try check(AudioDeviceStart(aggID, procID), "start aggregate device")

        // Output device changed (AirPods, HDMI…): exit so the Go side restarts us on the new device.
        var addr = address(kAudioHardwarePropertyDefaultSystemOutputDevice)
        AudioObjectAddPropertyListenerBlock(systemObject, &addr, queue) { [weak self] _, _ in
            fputs("lampa-tap: output device changed, exiting\n", stderr)
            self?.stopAndExit(0)
        }
    }

    private func handle(_ input: UnsafePointer<AudioBufferList>, nonInterleaved: Bool) {
        let abl = UnsafeMutableAudioBufferListPointer(UnsafeMutablePointer(mutating: input))
        guard let first = abl.first, first.mData != nil else { return }
        var mono: [Float]
        if nonInterleaved {
            let frames = Int(first.mDataByteSize) / 4
            mono = [Float](repeating: 0, count: frames)
            for buf in abl {
                guard let p = buf.mData?.assumingMemoryBound(to: Float.self) else { continue }
                for i in 0..<frames { mono[i] += p[i] }
            }
            let k = Float(abl.count)
            for i in 0..<frames { mono[i] /= k }
        } else {
            let ch = max(1, Int(first.mNumberChannels))
            let frames = Int(first.mDataByteSize) / 4 / ch
            let p = first.mData!.assumingMemoryBound(to: Float.self)
            mono = [Float](repeating: 0, count: frames)
            for i in 0..<frames {
                var s: Float = 0
                for c in 0..<ch { s += p[i * ch + c] }
                mono[i] = s / Float(ch)
            }
        }
        let data = mono.withUnsafeBufferPointer { Data(buffer: $0) }
        queue.async { [weak self] in self?.writeAll(data) }
    }

    private func writeAll(_ data: Data) {
        data.withUnsafeBytes { raw in
            var off = 0
            while off < raw.count {
                let n = Darwin.write(1, raw.baseAddress! + off, raw.count - off)
                if n < 0 && errno == EINTR { continue }
                if n <= 0 { stopAndExit(0) } // reader went away
                off += n
            }
        }
    }

    func stop() {
        guard !stopped else { return }
        stopped = true
        if let procID {
            AudioDeviceStop(aggID, procID)
            AudioDeviceDestroyIOProcID(aggID, procID)
        }
        if aggID != kAudioObjectUnknown { AudioHardwareDestroyAggregateDevice(aggID) }
        if tapID != kAudioObjectUnknown { AudioHardwareDestroyProcessTap(tapID) }
    }

    func stopAndExit(_ code: Int32) -> Never {
        stop()
        exit(code)
    }
}
```

```go
// Package light defines lamps and drives each one from its own goroutine.
package light

import (
	"context"
	"math"
)

type Kind string

const (
	KindWiZ      Kind = "wiz"
	KindYeelight Kind = "yeelight"
	KindVirtual  Kind = "virtual"
)

type Frame struct {
	R, G, B    uint8
	Brightness float64
}

type Info struct {
	ID, Name, Model, Role string
	Kind                  Kind
	MaxFPS                float64
}

type State struct{ Data map[string]any }

type Light interface {
	Info() Info
	Apply(ctx context.Context, f Frame) error
	Snapshot(ctx context.Context) (State, error)
	Restore(ctx context.Context, s State) error
	Close() error
}

type Status string

const (
	StatusOK         Status = "ok"
	StatusLost       Status = "lost"
	StatusConnecting Status = "connecting"
)

// Statuser is implemented by drivers that know their connection state.
type Statuser interface {
	Status() (Status, string)
}

const gamma = 2.0

// Split maps a perceptual frame onto a device with a minimum brightness level.
func Split(f Frame, floor float64) (r, g, b uint8, level float64) {
	p := math.Pow(math.Max(0, math.Min(1, f.Brightness)), gamma)
	r, g, b = f.R, f.G, f.B
	level = p
	if p < floor {
		level = floor
		s := p / floor
		r = uint8(math.Round(float64(r) * s))
		g = uint8(math.Round(float64(g) * s))
		b = uint8(math.Round(float64(b) * s))
	}
	if r == 0 && g == 0 && b == 0 {
		switch {
		case f.R >= f.G && f.R >= f.B:
			r = 1
		case f.G >= f.B:
			g = 1
		default:
			b = 1
		}
	}
	return r, g, b, level
}
```

```go
package light

import (
	"context"
	"sync"
	"time"
)

const defaultFPS = 20

type Stats struct {
	Status Status
	Note   string
	FPS    float64
	Errors int
	Last   Frame
}

// Runner owns one lamp: it keeps only the newest frame, rate-limits to
// Info().MaxFPS and skips frames identical to the last one sent.
type Runner struct {
	l    Light
	wake chan struct{}

	mu      sync.Mutex
	pending Frame
	has     bool
	last    Frame
	sentAny bool
	failed  bool
	errors  int
	sends   []time.Time
}

func NewRunner(l Light) *Runner {
	return &Runner{l: l, wake: make(chan struct{}, 1)}
}

func (r *Runner) Light() Light { return r.l }

func (r *Runner) Set(f Frame) {
	r.mu.Lock()
	r.pending, r.has = f, true
	r.mu.Unlock()
	select {
	case r.wake <- struct{}{}:
	default:
	}
}

func (r *Runner) Run(ctx context.Context) {
	fps := r.l.Info().MaxFPS
	if fps <= 0 {
		fps = defaultFPS
	}
	interval := time.Duration(float64(time.Second) / fps)
	var lastSend time.Time
	for {
		select {
		case <-ctx.Done():
			return
		case <-r.wake:
		}
		if wait := interval - time.Since(lastSend); wait > 0 {
			select {
			case <-ctx.Done():
				return
			case <-time.After(wait):
			}
		}
		r.mu.Lock()
		f, has := r.pending, r.has
		r.has = false
		skip := !has || (r.sentAny && !r.failed && f == r.last)
		r.mu.Unlock()
		if skip {
			continue
		}
		actx, cancel := context.WithTimeout(ctx, time.Second)
		err := r.l.Apply(actx, f)
		cancel()
		lastSend = time.Now()
		r.mu.Lock()
		if err != nil {
			r.failed = true
			r.errors++
		} else {
			r.failed, r.last, r.sentAny = false, f, true
			r.sends = append(r.sends, lastSend)
		}
		r.mu.Unlock()
	}
}

func (r *Runner) Stats() Stats {
	r.mu.Lock()
	defer r.mu.Unlock()
	cut := time.Now().Add(-time.Second)
	i := 0
	for i < len(r.sends) && r.sends[i].Before(cut) {
		i++
	}
	r.sends = r.sends[i:]
	s := Stats{FPS: float64(len(r.sends)), Errors: r.errors, Last: r.last, Status: StatusOK}
	if r.failed {
		s.Status = StatusLost
	}
	if st, ok := r.l.(Statuser); ok {
		s.Status, s.Note = st.Status()
	}
	return s
}
```

```go
// Package wiz drives Philips WiZ bulbs over their local UDP API.
package wiz

import (
	"context"
	"encoding/json"
	"errors"
	"math"
	"net"
	"strconv"
	"sync"
	"time"

	"lampa/internal/light"
)

const (
	Port        = 38899
	dimmingMin  = 0.10
	defaultFPS  = 20
	heartbeat   = time.Second
	lostTimeout = 3 * time.Second
)

type Bulb struct {
	info      light.Info
	conn      *net.UDPConn
	pilots    chan map[string]any
	done      chan struct{}
	closeOnce sync.Once
	hb, lost  time.Duration

	mu       sync.Mutex
	lastSeen time.Time
}

func Dial(ip string, info light.Info) (*Bulb, error) {
	return dial(net.JoinHostPort(ip, strconv.Itoa(Port)), info, heartbeat, lostTimeout)
}

func dial(addr string, info light.Info, hb, lost time.Duration) (*Bulb, error) {
	raddr, err := net.ResolveUDPAddr("udp", addr)
	if err != nil {
		return nil, err
	}
	conn, err := net.DialUDP("udp", nil, raddr)
	if err != nil {
		return nil, err
	}
	info.Kind = light.KindWiZ
	if info.MaxFPS <= 0 {
		info.MaxFPS = defaultFPS
	}
	b := &Bulb{info: info, conn: conn, pilots: make(chan map[string]any, 1),
		done: make(chan struct{}), hb: hb, lost: lost}
	go b.readLoop()
	go b.heartbeatLoop()
	return b, nil
}

func (b *Bulb) Info() light.Info { return b.info }

func (b *Bulb) send(method string, params map[string]any) error {
	msg, err := json.Marshal(map[string]any{"method": method, "params": params})
	if err != nil {
		return err
	}
	_, err = b.conn.Write(msg)
	return err
}

func (b *Bulb) readLoop() {
	buf := make([]byte, 4096)
	for {
		n, err := b.conn.Read(buf)
		if err != nil {
			select {
			case <-b.done:
				return
			default:
				continue // e.g. ICMP port unreachable while the bulb is off
			}
		}
		var msg struct {
			Method string         `json:"method"`
			Result map[string]any `json:"result"`
		}
		if json.Unmarshal(buf[:n], &msg) != nil {
			continue
		}
		b.mu.Lock()
		b.lastSeen = time.Now()
		b.mu.Unlock()
		if msg.Method == "getPilot" && msg.Result != nil {
			select {
			case b.pilots <- msg.Result:
			default:
			}
		}
	}
}

func (b *Bulb) heartbeatLoop() {
	t := time.NewTicker(b.hb)
	defer t.Stop()
	b.send("getPilot", map[string]any{})
	for {
		select {
		case <-b.done:
			return
		case <-t.C:
			b.send("getPilot", map[string]any{})
		}
	}
}

func (b *Bulb) Status() (light.Status, string) {
	b.mu.Lock()
	defer b.mu.Unlock()
	switch {
	case b.lastSeen.IsZero():
		return light.StatusConnecting, "ждём ответа"
	case time.Since(b.lastSeen) > b.lost:
		return light.StatusLost, "нет ответа"
	}
	return light.StatusOK, ""
}

func (b *Bulb) Apply(_ context.Context, f light.Frame) error {
	r, g, bl, lvl := light.Split(f, dimmingMin)
	dimming := int(math.Round(lvl * 100))
	dimming = max(10, min(100, dimming))
	return b.send("setPilot", map[string]any{"state": true, "r": r, "g": g, "b": bl, "dimming": dimming})
}

func (b *Bulb) Snapshot(ctx context.Context) (light.State, error) {
	select { // drop a stale heartbeat reply
	case <-b.pilots:
	default:
	}
	if err := b.send("getPilot", map[string]any{}); err != nil {
		return light.State{}, err
	}
	retry := time.NewTicker(300 * time.Millisecond)
	defer retry.Stop()
	for {
		select {
		case p := <-b.pilots:
			return light.State{Data: p}, nil
		case <-retry.C:
			b.send("getPilot", map[string]any{})
		case <-ctx.Done():
			return light.State{}, errors.New("wiz: no reply to getPilot")
		}
	}
}

func pick(src map[string]any, keys ...string) map[string]any {
	out := map[string]any{}
	for _, k := range keys {
		if v, ok := src[k]; ok {
			out[k] = v
		}
	}
	return out
}

func num(v any) float64 {
	f, _ := v.(float64)
	return f
}

func (b *Bulb) Restore(ctx context.Context, s light.State) error {
	d := s.Data
	if d == nil {
		return nil
	}
	var params map[string]any
	switch {
	case d["state"] == false:
		params = map[string]any{"state": false}
	case num(d["sceneId"]) > 0:
		params = pick(d, "sceneId", "speed", "dimming")
	case num(d["temp"]) > 0:
		params = pick(d, "temp", "dimming")
	default:
		params = pick(d, "r", "g", "b", "c", "w", "dimming")
	}
	if d["state"] != false {
		params["state"] = true
	}
	// UDP: repeat a few times so a single dropped packet doesn't leave the lamp wrong.
	var err error
	for i := 0; i < 3; i++ {
		err = b.send("setPilot", params)
		select {
		case <-ctx.Done():
			return err
		case <-time.After(50 * time.Millisecond):
		}
	}
	return err
}

func (b *Bulb) Close() error {
	var err error
	b.closeOnce.Do(func() {
		close(b.done)
		err = b.conn.Close()
	})
	return err
}
```

```go
	info.Kind = light.KindYeelight
	if !music {
		info.MaxFPS = fallbackFPS
	} else if info.MaxFPS <= 0 {
		info.MaxFPS = musicFPS
	}
	b := &Bulb{info: info, addr: addr, music: music, backoffMin: bmin, backoffMax: bmax,
		done: make(chan struct{}), status: light.StatusConnecting}
	go b.loop()
	return b
}

func (b *Bulb) Info() light.Info { return b.info }

func (b *Bulb) Status() (light.Status, string) {
	b.mu.Lock()
	defer b.mu.Unlock()
	return b.status, b.note
}

func (b *Bulb) setStatus(s light.Status, note string) {
	b.mu.Lock()
	b.status, b.note = s, note
	b.mu.Unlock()
}

func (b *Bulb) cmd(method string, params ...any) []byte {
	msg, _ := json.Marshal(map[string]any{"id": b.nextID.Add(1), "method": method, "params": params})
	return append(msg, '\r', '\n')
}

func (b *Bulb) stopped() bool {
	select {
	case <-b.done:
		return true
	default:
		return false
	}
}

func (b *Bulb) loop() {
	backoff := b.backoffMin
	for !b.stopped() {
		b.setStatus(light.StatusConnecting, "")
		conn, err := b.connect()
		if err != nil {
			b.setStatus(light.StatusLost, err.Error())
			select {
			case <-b.done:
				return
			case <-time.After(backoff):
			}
			backoff = min(backoff*2, b.backoffMax)
			continue
		}
		backoff = b.backoffMin
		note := ""
		if !b.music {
			note = "без music mode: ≈0,5 кадра/с"
		}
		b.mu.Lock()
		if b.stopped() { // shutdown raced with connect
			b.mu.Unlock()
			conn.Close()
			return
		}
		b.stream, b.status, b.note = conn, light.StatusOK, note
		b.mu.Unlock()
		io.Copy(io.Discard, conn) // returns when the bulb drops the connection
		b.mu.Lock()
		if b.stream == conn {
			b.stream = nil
		}
		b.mu.Unlock()
		conn.Close()
	}
}

// connect returns the connection frames are written to: the bulb's
// music-mode callback connection, or the control connection in fallback.
func (b *Bulb) connect() (net.Conn, error) {
	ctrl, err := net.DialTimeout("tcp", b.addr, dialTimeout)
	if err != nil {
		return nil, err
	}
	if !b.music {
		if _, err := ctrl.Write(b.cmd("set_power", "on", "smooth", 300)); err != nil {
			ctrl.Close()
			return nil, err
		}
		return ctrl, nil
	}
	defer ctrl.Close()
	localIP := ctrl.LocalAddr().(*net.TCPAddr).IP
	ln, err := net.ListenTCP("tcp", &net.TCPAddr{IP: localIP})
	if err != nil {
		return nil, err
	}
	defer ln.Close()
	port := ln.Addr().(*net.TCPAddr).Port
	if _, err := ctrl.Write(b.cmd("set_music", 1, localIP.String(), port)); err != nil {
		return nil, err
	}
	ctrl.SetReadDeadline(time.Now().Add(dialTimeout))
	if line, err := bufio.NewReader(ctrl).ReadString('\n'); err != nil {
		return nil, fmt.Errorf("set_music: %w", err)
	} else if strings.Contains(line, `"error"`) {
		return nil, fmt.Errorf("set_music отклонён: %s", strings.TrimSpace(line))
	}
	ln.SetDeadline(time.Now().Add(acceptWait))
	conn, err := ln.Accept()
	if err != nil {
		return nil, fmt.Errorf("лампа не подключилась в music mode: %w", err)
	}
	if _, err := conn.Write(b.cmd("set_power", "on", "sudden", 30)); err != nil {
		conn.Close()
		return nil, err
	}
	return conn, nil
}

func (b *Bulb) Apply(_ context.Context, f light.Frame) error {
	b.mu.Lock()
	conn := b.stream
	b.mu.Unlock()
	if conn == nil {
		return errNotConnected
	}
	r, g, bl, lvl := light.Split(f, 0.01)
	rgb := int(r)<<16 | int(g)<<8 | int(bl)
	bright := max(1, min(100, int(math.Round(lvl*100))))
	effect, dur := "sudden", 30
	if !b.music {
		effect, dur = "smooth", 900
	}
	msg := append(b.cmd("set_rgb", rgb, effect, dur), b.cmd("set_bright", bright, effect, dur)...)
	conn.SetWriteDeadline(time.Now().Add(writeTimeout))
	if _, err := conn.Write(msg); err != nil {
		conn.Close() // loop notices and reconnects
		return err
	}
	return nil
}

var snapshotProps = []string{"power", "bright", "rgb", "ct", "color_mode"}

func (b *Bulb) Snapshot(ctx context.Context) (light.State, error) {
	var d net.Dialer
	ctrl, err := d.DialContext(ctx, "tcp", b.addr)
	if err != nil {
		return light.State{}, err
	}
	defer ctrl.Close()
	if dl, ok := ctx.Deadline(); ok {
		ctrl.SetDeadline(dl)
	}
	params := make([]any, len(snapshotProps))
	for i, p := range snapshotProps {
		params[i] = p
	}
	req := b.cmd("get_prop", params...)
	var sent struct{ ID int64 `json:"id"` }
	json.Unmarshal(req, &sent)
	if _, err := ctrl.Write(req); err != nil {
		return light.State{}, err
	}
	rd := bufio.NewReader(ctrl)
	for {
		line, err := rd.ReadString('\n')
		if err != nil {
			return light.State{}, fmt.Errorf("get_prop: %w", err)
		}
		var resp struct {
			ID     int64    `json:"id"`
			Result []string `json:"result"`
		}
		if json.Unmarshal([]byte(line), &resp) != nil || resp.ID != sent.ID {
			continue // notification or unrelated reply
		}
		data := map[string]any{}
		for i, p := range snapshotProps {
			if i < len(resp.Result) {
				data[p] = resp.Result[i]
			}
		}
		return light.State{Data: data}, nil
	}
}

// Restore stops streaming, leaves music mode and puts the bulb back.
func (b *Bulb) Restore(ctx context.Context, s light.State) error {
	b.shutdown()
	var d net.Dialer
	ctrl, err := d.DialContext(ctx, "tcp", b.addr)
	if err != nil {
		return err
	}
	defer ctrl.Close()
	ctrl.SetDeadline(time.Now().Add(2 * time.Second))
	var msgs [][]byte
	if b.music {
		msgs = append(msgs, b.cmd("set_music", 0))
	}
	str := func(k string) string { v, _ := s.Data[k].(string); return v }
	atoi := func(k string) int { n, _ := strconv.Atoi(str(k)); return n }
	if s.Data != nil {
		if str("power") == "off" {
			msgs = append(msgs, b.cmd("set_power", "off", "smooth", 500))
		} else {
			switch str("color_mode") {
			case "1":
				msgs = append(msgs, b.cmd("set_rgb", atoi("rgb"), "smooth", 500))
			case "2":
				msgs = append(msgs, b.cmd("set_ct_abx", atoi("ct"), "smooth", 500))
			}
			if br := atoi("bright"); br > 0 {
				msgs = append(msgs, b.cmd("set_bright", br, "smooth", 500))
			}
		}
	}
	for _, m := range msgs {
		if _, err := ctrl.Write(m); err != nil {
			return err
		}
	}
	rd := bufio.NewReader(ctrl) // wait for replies so commands land before we close
	for range msgs {
		if _, err := rd.ReadString('\n'); err != nil {
			break
		}
	}
	return nil
}

// shutdown stops the connect loop without waiting for an in-flight dial
// (an offline bulb must not delay exit); loop() re-checks stopped() under mu.
func (b *Bulb) shutdown() {
	b.stopOnce.Do(func() { close(b.done) })
	b.mu.Lock()
	if b.stream != nil {
		b.stream.Close()
		b.stream = nil
	}
	b.mu.Unlock()
}

func (b *Bulb) Close() error {
	b.shutdown()
	return nil
}
```

```go
package discovery

import (
	"context"
	"encoding/json"
	"net"
	"net/url"
	"slices"
	"strings"
	"time"

	"lampa/internal/light"

type Found struct {
	Kind    light.Kind
	IP      string
	ID      string
	Model   string
	Name    string
	Music   bool
	Support []string
}

func suffix(s string, n int) string {
	if len(s) <= n {
		return s
	}
	return s[len(s)-n:]
}

func ParseWiZ(ip string, data []byte) (Found, bool) {
	var msg struct {
		Method string `json:"method"`
		Result struct {
			Mac        string `json:"mac"`
			ModuleName string `json:"moduleName"`
		} `json:"result"`
	}
	if json.Unmarshal(data, &msg) != nil || msg.Method != "getSystemConfig" || msg.Result.Mac == "" {
		return Found{}, false
	}
	return Found{Kind: light.KindWiZ, IP: ip, ID: msg.Result.Mac, Model: msg.Result.ModuleName,
		Name: "wiz-" + suffix(msg.Result.Mac, 4)}, true
}

func ParseYeelight(data []byte) (Found, bool) {
	lines := strings.Split(string(data), "\r\n")
	if len(lines) == 0 || !strings.HasPrefix(lines[0], "HTTP/1.1 200") {
		return Found{}, false
	}
	h := map[string]string{}
	for _, l := range lines[1:] {
		if k, v, ok := strings.Cut(l, ":"); ok {
			h[strings.ToLower(strings.TrimSpace(k))] = strings.TrimSpace(v)
		}
	}
	u, err := url.Parse(h["location"])
	if err != nil || u.Scheme != "yeelight" || h["id"] == "" {
		return Found{}, false
	}
	support := strings.Fields(h["support"])
	f := Found{Kind: light.KindYeelight, IP: u.Hostname(), ID: h["id"], Model: h["model"],
		Name: h["name"], Support: support, Music: slices.Contains(support, "set_music")}
	if f.Name == "" {
		f.Name = "yeelight-" + suffix(f.ID, 4)
	}
	return f, true
}

const ssdpSearch = "M-SEARCH * HTTP/1.1\r\nHOST: 239.255.255.250:1982\r\nMAN: \"ssdp:discover\"\r\nST: wifi_bulb\r\n\r\n"

func Discover(ctx context.Context, wait time.Duration) ([]Found, error) {
	conn, err := net.ListenUDP("udp4", &net.UDPAddr{})
	if err != nil {
		return nil, err
	}
	defer conn.Close()
	wizAddr := &net.UDPAddr{IP: net.IPv4bcast, Port: 38899}
	ssdpAddr := &net.UDPAddr{IP: net.IPv4(239, 255, 255, 250), Port: 1982}
	probe := func() {
		conn.WriteToUDP([]byte(`{"method":"getSystemConfig","params":{}}`), wizAddr)
		conn.WriteToUDP([]byte(ssdpSearch), ssdpAddr)
	}
	probe()
	deadline := time.Now().Add(wait)
	if dl, ok := ctx.Deadline(); ok && dl.Before(deadline) {
		deadline = dl
	}
	resent := false
	seen := map[string]Found{}
	buf := make([]byte, 4096)
	for time.Now().Before(deadline) {
		if !resent && time.Until(deadline) < wait/2 {
			probe() // UDP is lossy: ask twice
			resent = true
		}
		conn.SetReadDeadline(time.Now().Add(100 * time.Millisecond))
		n, from, err := conn.ReadFromUDP(buf)
		if ctx.Err() != nil {
			break
		}
		if err != nil {
			continue
		}
		f, ok := ParseWiZ(from.IP.String(), buf[:n])
		if !ok {
			f, ok = ParseYeelight(buf[:n])
		}
		if ok {
			seen[string(f.Kind)+"/"+f.ID] = f
		}
	}
	out := make([]Found, 0, len(seen))
	for _, f := range seen {
		out = append(out, f)
	}
	slices.SortFunc(out, func(a, b Found) int {
		if c := strings.Compare(string(a.Kind), string(b.Kind)); c != 0 {
			return c
		}
		return strings.Compare(a.Name, b.Name)
	})
	return out, nil
}
```

```go
// Package config loads and saves ~/.config/lampa/config.toml.
package config

import (
	"bytes"
	"errors"
	"io/fs"
	"os"
	"path/filepath"

	"github.com/BurntSushi/toml"

	"lampa/internal/discovery"
)

type Lamp struct {
	Kind     string  `toml:"kind"`
	IP       string  `toml:"ip"`
	ID       string  `toml:"id"`
	Name     string  `toml:"name"`
	Model    string  `toml:"model,omitempty"`
	Role     string  `toml:"role,omitempty"`
	Music    bool    `toml:"music"`
	MaxFPS   float64 `toml:"max_fps,omitempty"`
	Disabled bool    `toml:"disabled,omitempty"`
}

type Config struct {
	Mode         string  `toml:"mode"`
	Palette      string  `toml:"palette"`
	Sensitivity  float64 `toml:"sensitivity"`
	LatencyMS    int     `toml:"latency_ms"`
	SourceBundle string  `toml:"source_bundle,omitempty"`
	Lamps        []Lamp  `toml:"lamp"`
}

func Default() Config {
	return Config{Mode: "pulse", Palette: "sunset", Sensitivity: 0.5, LatencyMS: 60}
}

func DefaultPath() string {
	home, err := os.UserHomeDir()
	if err != nil {
		return "config.toml"
	}
	return filepath.Join(home, ".config", "lampa", "config.toml")
}

func Load(path string) (Config, error) {
	c := Default()
	_, err := toml.DecodeFile(path, &c)
	if errors.Is(err, fs.ErrNotExist) {
		return Default(), nil
	}
	return c, err
}

func Save(path string, c Config) error {
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		return err
	}
	var buf bytes.Buffer
	if err := toml.NewEncoder(&buf).Encode(c); err != nil {
		return err
	}
	tmp := path + ".tmp"
	if err := os.WriteFile(tmp, buf.Bytes(), 0o644); err != nil {
		return err
	}
	return os.Rename(tmp, path)
}

func (c *Config) Merge(found []discovery.Found) (added int) {
	for _, f := range found {
		known := false
		for i := range c.Lamps {
			l := &c.Lamps[i]
			if l.Kind == string(f.Kind) && l.ID == f.ID {
				l.IP, l.Model, l.Music = f.IP, f.Model, f.Music
				known = true
				break
			}
		}
		if !known {
			c.Lamps = append(c.Lamps, Lamp{Kind: string(f.Kind), IP: f.IP, ID: f.ID, Name: f.Name, Model: f.Model, Music: f.Music})
			added++
		}
	}
	return added
}
```

```go
// Package modes turns musical features into per-lamp colors.
package modes

import (
	"math"
	"strings"
	"time"

	"lampa/internal/analysis"
	"lampa/internal/color"
	"lampa/internal/light"
)

type Target struct {
	Color      color.OKLCH
	Brightness float64
}

type Scene []Target

type Context struct {
	Palette color.Palette
	Lights  []light.Info
}

type Mode interface {
	Name() string
	Update(f analysis.Features, c Context) Scene
	Envelope() (attack, release time.Duration)
}

func All() []Mode {
	return []Mode{&Pulse{}, &Spectrum{}, &Chase{}, &Ambient{}, &Drop{}}
}

func Index(name string) int {
	for i, m := range All() {
		if strings.EqualFold(m.Name(), name) {
			return i
		}
	}
	return 0
}

const minConfidence = 0.4

// hit is the rhythmic trigger: beats when the tempo is trusted, kicks otherwise.
func hit(f analysis.Features) bool {
	if f.Confidence >= minConfidence {
		return f.Beat
	}
	return f.OnsetLow.Hit
}

func clamp01(x float64) float64 { return math.Max(0, math.Min(1, x)) }

// clock turns feature timestamps into a bounded frame delta.
type clock struct {
	last    time.Duration
	started bool
}

func (c *clock) dt(now time.Duration) float64 {
	if !c.started {
		c.started, c.last = true, now
		return 0
	}
	d := (now - c.last).Seconds()
	c.last = now
	return math.Max(0, math.Min(0.1, d))
}
```

```go
package modes

import (
	"time"

	"lampa/internal/analysis"
)

// Pulse: every lamp flashes on the beat; the palette steps forward each bar.
type Pulse struct{ hue float64 }

func (p *Pulse) Name() string                    { return "Pulse" }
func (p *Pulse) Envelope() (time.Duration, time.Duration) { return 0, 250 * time.Millisecond }

func (p *Pulse) Update(f analysis.Features, c Context) Scene {
	h := hit(f)
	if h {
		if f.Confidence < minConfidence {
			p.hue += 0.03
		} else if f.BarPhase < 0.125 {
			p.hue += 0.125
		}
	}
	s := make(Scene, len(c.Lights))
	base := 0.12 + 0.25*f.Loudness
	for i := range s {
		b := base
		if h {
			b = 1
		}
		s[i] = Target{Color: c.Palette.At(p.hue + 0.06*float64(i)), Brightness: clamp01(b)}
	}
	return s
}
```

```go
package modes

import (
	"time"

	"lampa/internal/analysis"
)

var groupNames = []string{"sub", "bass", "lowmid", "mid", "high"}

// default assignment for lamps without a role: bass, mid, high, bass, …
var defaultGroups = []int{1, 3, 4}

func groupValue(g analysis.Groups, i int) float64 {
	return [...]float64{g.Sub, g.Bass, g.LowMid, g.Mid, g.High}[i]
}

// Spectrum: each lamp follows one frequency group.
type Spectrum struct {
	clk  clock
	hue  float64
}

func (m *Spectrum) Name() string                    { return "Spectrum" }
func (m *Spectrum) Envelope() (time.Duration, time.Duration) { return 0, 150 * time.Millisecond }

func (m *Spectrum) Update(f analysis.Features, c Context) Scene {
	m.hue += m.clk.dt(f.Time) * 0.01
	s := make(Scene, len(c.Lights))
	for i, l := range c.Lights {
		g := defaultGroups[i%len(defaultGroups)]
		for j, n := range groupNames {
			if l.Role == n {
				g = j
			}
		}
		s[i] = Target{
			Color:      c.Palette.At(m.hue + float64(g)/float64(len(groupNames))),
			Brightness: clamp01(groupValue(f.Groups, g)),
		}
	}
	return s
}
```

```go
// Chase: the light jumps to the next lamp on every beat.
type Chase struct {
	pos, step int
}

func (m *Chase) Name() string                    { return "Chase" }
func (m *Chase) Envelope() (time.Duration, time.Duration) { return 0, 400 * time.Millisecond }

func (m *Chase) Update(f analysis.Features, c Context) Scene {
	n := len(c.Lights)
	if n == 0 {
		return Scene{}
	}
	if hit(f) {
		m.pos = (m.pos + 1) % n
		m.step++
	}
	s := make(Scene, n)
	t := float64(m.step) * 0.1
	for i := range s {
		if i == m.pos {
			s[i] = Target{Color: c.Palette.At(t), Brightness: 1}
		} else {
			s[i] = Target{Color: c.Palette.At(t - 0.1), Brightness: 0.06}
		}
	}
	return s
}
```

```go
package modes

import (
	"math"
	"time"

	"lampa/internal/analysis"
)

// Ambient: slow drift; tempo sets the speed, loudness the saturation.
type Ambient struct {
	clk  clock
	t    float64
	loud float64
}

func (m *Ambient) Name() string                    { return "Ambient" }
func (m *Ambient) Envelope() (time.Duration, time.Duration) { return 1500 * time.Millisecond, 1500 * time.Millisecond }

func (m *Ambient) Update(f analysis.Features, c Context) Scene {
	dt := m.clk.dt(f.Time)
	speed := 0.02
	if f.BPM > 0 {
		speed *= f.BPM / 120
	}
	m.t += dt * speed
	m.loud += (f.Loudness - m.loud) * (1 - math.Exp(-dt/1.0))
	n := len(c.Lights)
	s := make(Scene, n)
	for i := range s {
		col := c.Palette.At(m.t + 0.3*float64(i)/float64(n))
		col.C *= 0.5 + 0.5*m.loud
		s[i] = Target{Color: col, Brightness: clamp01(0.35 + 0.4*m.loud)}
	}
	return s
}
```

```go
package modes

import (
	"time"

	"lampa/internal/analysis"
	"lampa/internal/color"
)

const (
	flashLen = 150 * time.Millisecond
	glowLen  = 4 * time.Second
)

// Drop: tension accumulates during a build; a strong kick after it fires
// a white flash and a burst of color.
type Drop struct {
	clk        clock
	charge     float64
	hue        float64
	flashUntil time.Duration
	glowUntil  time.Duration
}

func (m *Drop) Name() string                    { return "Drop" }
func (m *Drop) Envelope() (time.Duration, time.Duration) { return 0, 300 * time.Millisecond }

func (m *Drop) Update(f analysis.Features, c Context) Scene {
	dt := m.clk.dt(f.Time)
	if f.Build > 0.2 {
		m.charge += f.Build * dt * 0.4
	} else {
		m.charge -= dt * 0.2
	}
	m.charge = clamp01(m.charge)
	if m.charge > 0.5 && f.OnsetLow.Hit && f.OnsetLow.Strength > 0.3 {
		m.flashUntil, m.glowUntil = f.Time+flashLen, f.Time+glowLen
		m.hue += 0.37
		m.charge = 0
	}
	s := make(Scene, len(c.Lights))
	for i := range s {
		switch {
		case f.Time < m.flashUntil:
			s[i] = Target{Color: color.OKLCH{L: 1}, Brightness: 1}
		case f.Time < m.glowUntil:
			b := 0.5
			if hit(f) {
				b = 1
			}
			s[i] = Target{Color: c.Palette.At(m.hue + 0.1*float64(i)), Brightness: b}
		default:
			col := c.Palette.At(m.hue + 0.05*float64(i))
			col.C *= 0.3 + 0.7*m.charge
			b := 0.15 + 0.6*m.charge
			if m.charge > 0.3 && f.OnsetHigh.Hit {
				b += 0.2
			}
			s[i] = Target{Color: col, Brightness: clamp01(b)}
		}
	}
	return s
}
```

```go
// Package engine connects audio, analysis, modes and lamps.
package engine

import (
	"context"
	"sync"
	"time"

	"lampa/internal/analysis"
	"lampa/internal/audio"
	"lampa/internal/color"
	"lampa/internal/light"
	"lampa/internal/modes"
)

const (
	HistoryLen      = 188
	snapshotTimeout = 2 * time.Second
	restoreTimeout  = 2 * time.Second
	silenceAfter    = time.Second
	maxLatency      = 200 * time.Millisecond
)

type Options struct {
	Lights      []light.Light
	Mode        int
	Palette     int
	Sensitivity float64
	Latency     time.Duration
}

type LampView struct {
	Info       light.Info
	Stats      light.Stats
	Color      color.RGB
	Brightness float64
}

type Marker struct{ Beat, Low, High bool }

type View struct {
	Features    analysis.Features
	Mode        string
	Palette     string
	Sensitivity float64
	Latency     time.Duration
	Paused      bool
	AudioAlive  bool
	Lamps       []LampView
	History     []Marker
}

type lamp struct {
	l      light.Light
	r      *light.Runner
	env    color.Envelope
	state  light.State
	color  color.RGB
	bright float64
}

type Engine struct {
	mu          sync.Mutex
	lamps       []*lamp
	modes       []modes.Mode
	mode        int
	palette     int
	sensitivity float64
	latency     time.Duration
	paused      bool
	an          *analysis.Analyzer
	last        analysis.Features
	lastTime    time.Duration
	history     []Marker
	lastAudio   time.Time
	runCtx      context.Context
	cancelRuns  context.CancelFunc
	wg          sync.WaitGroup
}

func New(o Options) *Engine {
	if o.Sensitivity == 0 {
		o.Sensitivity = 0.5
	}
	e := &Engine{modes: modes.All(), sensitivity: o.Sensitivity, latency: o.Latency}
	e.mode = o.Mode
	if e.mode < 0 || e.mode >= len(e.modes) {
		e.mode = 0
	}
	e.palette = o.Palette % len(color.Palettes)
	for _, l := range o.Lights {
		e.lamps = append(e.lamps, &lamp{l: l, r: light.NewRunner(l)})
	}
	e.runCtx, e.cancelRuns = context.WithCancel(context.Background())
	return e
}

func snapshot(ctx context.Context, lp *lamp) {
	sctx, cancel := context.WithTimeout(ctx, snapshotTimeout)
	defer cancel()
	if s, err := lp.l.Snapshot(sctx); err == nil {
		lp.state = s
	}
}

func (e *Engine) startRunner(lp *lamp) {
	e.wg.Add(1)
	go func() {
		defer e.wg.Done()
		lp.r.Run(e.runCtx)
	}()
}

func (e *Engine) Start(ctx context.Context) {
	e.mu.Lock()
	lamps := append([]*lamp(nil), e.lamps...)
	e.mu.Unlock()
	var wg sync.WaitGroup
	for _, lp := range lamps {
		wg.Add(1)
		go func() { defer wg.Done(); snapshot(ctx, lp) }()
	}
	wg.Wait()
	for _, lp := range lamps {
		e.startRunner(lp)
	}
}

func (e *Engine) AddLight(l light.Light) {
	lp := &lamp{l: l, r: light.NewRunner(l)}
	snapshot(context.Background(), lp)
	e.mu.Lock()
	e.lamps = append(e.lamps, lp)
	e.mu.Unlock()
	e.startRunner(lp)
}

func (e *Engine) Run(ctx context.Context, blocks <-chan audio.Block) {
	tick := time.NewTicker(100 * time.Millisecond)
	defer tick.Stop()
	for {
		select {
		case <-ctx.Done():
			return
		case b := <-blocks:
			e.mu.Lock()
			e.lastAudio = time.Now()
			e.mu.Unlock()
			e.Process(b)
		case <-tick.C:
			e.mu.Lock()
			silent := time.Since(e.lastAudio) > silenceAfter
			rate := 48000
			if e.an != nil {
				rate = e.an.Rate()
			}
			e.mu.Unlock()
			if silent { // keep time moving so lamps fade to their resting state
				e.Process(audio.Block{Rate: rate, Samples: make([]float32, rate/10)})
			}
		}
	}
}

func (e *Engine) Process(b audio.Block) {
	e.mu.Lock()
	defer e.mu.Unlock()
	if e.an == nil || e.an.Rate() != b.Rate {
		e.an = analysis.New(analysis.Config{SampleRate: b.Rate, Sensitivity: e.sensitivity, Latency: e.latency})
		e.lastTime = 0
	}
	for _, f := range e.an.Push(b.Samples) {
		e.step(f)
	}
}

// step must be called with e.mu held.
func (e *Engine) step(f analysis.Features) {
	m := e.modes[e.mode]
	infos := make([]light.Info, len(e.lamps))
	for i, lp := range e.lamps {
		infos[i] = lp.l.Info()
	}
	scene := m.Update(f, modes.Context{Palette: color.Palettes[e.palette], Lights: infos})
	att, rel := m.Envelope()
	dt := f.Time - e.lastTime
	if dt <= 0 || dt > 100*time.Millisecond {
		dt = 10 * time.Millisecond
	}
	e.lastTime = f.Time
	for i, lp := range e.lamps {
		if i >= len(scene) {
			break
		}
		lp.env.Attack, lp.env.Release = att, rel
		lp.bright = lp.env.Step(scene[i].Brightness, dt)
		lp.color = scene[i].Color.ToRGB()
		if !e.paused {
			r, g, b := lp.color.Bytes()
			lp.r.Set(light.Frame{R: r, G: g, B: b, Brightness: lp.bright})
		}
	}
	e.last = f
	e.history = append(e.history, Marker{Beat: f.Beat, Low: f.OnsetLow.Hit, High: f.OnsetHigh.Hit})
	if len(e.history) > HistoryLen {
		e.history = e.history[len(e.history)-HistoryLen:]
	}
}

func (e *Engine) View() View {
	e.mu.Lock()
	defer e.mu.Unlock()
	v := View{
		Features:    e.last,
		Mode:        e.modes[e.mode].Name(),
		Palette:     color.Palettes[e.palette].Name,
		Sensitivity: e.sensitivity,
		Latency:     e.latency,
		Paused:      e.paused,
		AudioAlive:  !e.lastAudio.IsZero() && time.Since(e.lastAudio) <= silenceAfter,
		History:     append([]Marker(nil), e.history...),
	}
	for _, lp := range e.lamps {
		v.Lamps = append(v.Lamps, LampView{Info: lp.l.Info(), Stats: lp.r.Stats(), Color: lp.color, Brightness: lp.bright})
	}
	return v
}

func (e *Engine) SetMode(i int) {
	e.mu.Lock()
	defer e.mu.Unlock()
	if i >= 0 && i < len(e.modes) {
		e.mode = i
	}
}

func (e *Engine) NextPalette() {
	e.mu.Lock()
	e.palette = (e.palette + 1) % len(color.Palettes)
	e.mu.Unlock()
}

func (e *Engine) AdjustSensitivity(d float64) {
	e.mu.Lock()
	defer e.mu.Unlock()
	e.sensitivity = max(0, min(1, e.sensitivity+d))
	e.sensitivity = float64(int(e.sensitivity*100+0.5)) / 100 // keep steps clean for display
	if e.an != nil {
		e.an.SetSensitivity(e.sensitivity)
	}
}

func (e *Engine) AdjustLatency(d time.Duration) {
	e.mu.Lock()
	defer e.mu.Unlock()
	e.latency = max(-maxLatency, min(maxLatency, e.latency+d))
	if e.an != nil {
		e.an.SetLatency(e.latency)
	}
}

func (e *Engine) TogglePause() {
	e.mu.Lock()
	e.paused = !e.paused
	e.mu.Unlock()
}

func (e *Engine) Close(ctx context.Context) {
	e.cancelRuns()
	e.wg.Wait()
	e.mu.Lock()
	lamps := append([]*lamp(nil), e.lamps...)
	e.mu.Unlock()
	var wg sync.WaitGroup
	for _, lp := range lamps {
		wg.Add(1)
		go func() {
			defer wg.Done()
			rctx, cancel := context.WithTimeout(ctx, restoreTimeout)
			defer cancel()
			lp.l.Restore(rctx, lp.state)
			lp.l.Close()
		}()
	}
	wg.Wait()
}
```

```go
import (
```

```go
	"fmt"
	"time"

	tea "github.com/charmbracelet/bubbletea"

	"lampa/internal/audio"
	"lampa/internal/engine"

type Controller interface {
	View() engine.View
	SetMode(int)
	NextPalette()
	AdjustSensitivity(float64)
	AdjustLatency(time.Duration)
	TogglePause()
	Sources(ctx context.Context) ([]audio.App, error)
	SetSource(app *audio.App)
	SourceName() string
	Discover(ctx context.Context) (added int, err error)
}

type tickMsg time.Time

type appsMsg struct {
	apps []audio.App
	err  error
}

type discoveredMsg struct {
	added int
	err   error
}

type Model struct {
	c          Controller
	v          engine.View
	width      int
	picking    bool
	apps       []audio.App
	cursor     int
	status     string
}

func New(c Controller) Model {
	return Model{c: c, v: c.View(), width: 80}
}

func tick() tea.Cmd {
	return tea.Tick(33*time.Millisecond, func(t time.Time) tea.Msg { return tickMsg(t) })
}

func (m Model) Init() tea.Cmd { return tick() }

func (m Model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tickMsg:
		m.v = m.c.View()
		return m, tick()
	case tea.WindowSizeMsg:
		m.width = msg.Width
	case appsMsg:
		if msg.err != nil {
			m.status = "источники: " + msg.err.Error()
			return m, nil
		}
		m.picking, m.apps, m.cursor = true, msg.apps, 0
	case discoveredMsg:
		if msg.err != nil {
			m.status = "поиск: " + msg.err.Error()
		} else {
			m.status = fmt.Sprintf("найдено новых ламп: %d", msg.added)
		}
	case tea.KeyMsg:
		if m.picking {
			return m.updatePicker(msg)
		}
		return m.updateKeys(msg)
	}
	return m, nil
}

func (m Model) updatePicker(k tea.KeyMsg) (tea.Model, tea.Cmd) {
	switch k.String() {
	case "up", "k":
		m.cursor = max(0, m.cursor-1)
	case "down", "j":
		m.cursor = min(len(m.apps), m.cursor+1)
	case "enter":
		if m.cursor == 0 {
			m.c.SetSource(nil)
		} else {
			app := m.apps[m.cursor-1]
			m.c.SetSource(&app)
		}
		m.picking = false
	case "esc", "q":
		m.picking = false
	}
	return m, nil
}

func (m Model) updateKeys(k tea.KeyMsg) (tea.Model, tea.Cmd) {
	switch s := k.String(); s {
	case "q", "ctrl+c":
		return m, tea.Quit
	case "1", "2", "3", "4", "5":
		m.c.SetMode(int(s[0] - '1'))
	case "p":
		m.c.NextPalette()
	case "right":
		m.c.AdjustSensitivity(0.05)
	case "left":
		m.c.AdjustSensitivity(-0.05)
	case "up":
		m.c.AdjustLatency(10 * time.Millisecond)
	case "down":
		m.c.AdjustLatency(-10 * time.Millisecond)
	case " ":
		m.c.TogglePause()
	case "s":
		c := m.c
		return m, func() tea.Msg {
			ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
			defer cancel()
			apps, err := c.Sources(ctx)
			return appsMsg{apps, err}
		}
	case "d":
		m.status = "ищу лампы…"
		c := m.c
		return m, func() tea.Msg {
			ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
			defer cancel()
			n, err := c.Discover(ctx)
			return discoveredMsg{n, err}
		}
	}
	return m, nil
```

```go
import (

var (
	frameStyle = lipgloss.NewStyle().Border(lipgloss.RoundedBorder()).Padding(0, 1)
	dim        = lipgloss.NewStyle().Foreground(lipgloss.Color("8"))
	accent     = lipgloss.NewStyle().Foreground(lipgloss.Color("13")).Bold(true)
	warn       = lipgloss.NewStyle().Foreground(lipgloss.Color("9"))
	ok         = lipgloss.NewStyle().Foreground(lipgloss.Color("10"))
	bandColors = []string{"#ff5f5f", "#ff875f", "#ffaf5f", "#ffd75f", "#d7ff5f", "#87ff87", "#5fffd7", "#5fd7ff", "#87afff", "#af87ff"}
)

const levels = " ▁▂▃▄▅▆▇█"

func spectrumRows(bands []float64, rows int) []string {
	out := make([]string, rows)
	steps := 8 * rows
	for r := 0; r < rows; r++ {
		var b strings.Builder
		floor := (rows - 1 - r) * 8 // top row first
		for _, v := range bands {
			lvl := int(v*float64(steps) + 0.5)
			cell := max(0, min(8, lvl-floor))
			b.WriteRune([]rune(levels)[cell])
		}
		out[r] = b.String()
	}
	return out
}

func (m Model) View() string {
	if m.picking {
		return frameStyle.Render(m.pickerView())
	}
	v := m.v
	inner := max(40, m.width-4)
	var b strings.Builder

	beat := dim.Render("○")
	if v.Features.BPM > 0 && v.Features.BeatPhase < 0.15 {
		beat = accent.Render("●")
	}
	bpm := "— BPM"
	if v.Features.BPM > 0 {
		bpm = fmt.Sprintf("%.0f BPM", v.Features.BPM)
	}
	b.WriteString(fmt.Sprintf("%s  %s · %s %s   %s\n\n",
		accent.Render("lampa"), v.Mode, bpm, beat, dim.Render(m.c.SourceName())))

	// spectrum: each band repeated to fill the width
	per := max(1, inner/len(v.Features.Bands))
	for _, row := range spectrumRows(v.Features.Bands[:], 4) {
		for i, ch := range []rune(row) {
			st := lipgloss.NewStyle().Foreground(lipgloss.Color(bandColors[i*len(bandColors)/len(v.Features.Bands)]))
			b.WriteString(st.Render(strings.Repeat(string(ch), per)))
		}
		b.WriteByte('\n')
	}

	// beat / onset ribbon over the last ~2 s
	b.WriteString(ribbon(v.History, inner))
	b.WriteString("\n\n")

	// lamp tiles
	var tiles []string
	for _, l := range v.Lamps {
		tiles = append(tiles, tile(l))
	}
	if len(tiles) == 0 {
		b.WriteString(warn.Render("ламп нет — нажми d, чтобы найти"))
	} else {
		b.WriteString(lipgloss.JoinHorizontal(lipgloss.Top, tiles...))
	}
	b.WriteString("\n\n")

	state := []string{
		"палитра " + v.Palette,
		fmt.Sprintf("чувств. %.2f", v.Sensitivity),
		fmt.Sprintf("задержка %+d мс", v.Latency.Milliseconds()),
	}
	if v.Paused {
		state = append(state, warn.Render("пауза"))
	}
	if !v.AudioAlive {
		state = append(state, warn.Render("нет звука"))
	}
	b.WriteString(strings.Join(state, " · "))
	if m.status != "" {
		b.WriteString("\n" + m.status)
	}
	b.WriteString("\n" + dim.Render("[1-5] режим  [p] палитра  [←→] чувств.  [↑↓] задержка  [s] источник  [d] поиск  [space] пауза  [q] выход"))
	return frameStyle.Render(b.String())
}

func ribbon(h []engine.Marker, width int) string {
	if len(h) == 0 {
		return ""
	}
	var top, bottom strings.Builder
	for c := 0; c < width; c++ {
		lo := c * len(h) / width
		hi := max(lo+1, (c+1)*len(h)/width)
		var beat, low, high bool
		for _, mk := range h[lo:min(hi, len(h))] {
			beat, low, high = beat || mk.Beat, low || mk.Low, high || mk.High
		}
		switch {
		case beat:
			top.WriteString(accent.Render("┃"))
		default:
			top.WriteString(" ")
		}
		switch {
		case low:
			bottom.WriteString(warn.Render("█"))
		case high:
			bottom.WriteString(ok.Render("▪"))
		default:
			bottom.WriteString(dim.Render("·"))
		}
	}
	return top.String() + "\n" + bottom.String()
}

func tile(l engine.LampView) string {
	c := l.Color
	k := l.Brightness
	hex := fmt.Sprintf("#%02x%02x%02x", uint8(c.R*k*255), uint8(c.G*k*255), uint8(c.B*k*255))
	swatch := lipgloss.NewStyle().Background(lipgloss.Color(hex)).Width(14).Height(2).Render("")
	status := ok.Render(fmt.Sprintf("%2.0f fps ✓", l.Stats.FPS))
	switch l.Stats.Status {
	case light.StatusLost:
		status = warn.Render("потеряна")
	case light.StatusConnecting:
		status = dim.Render("подключаюсь…")
	}
	lines := []string{swatch, l.Info.Name, status}
	if l.Stats.Note != "" {
		lines = append(lines, dim.Render(l.Stats.Note))
	}
	return lipgloss.NewStyle().MarginRight(2).Width(16).Render(strings.Join(lines, "\n"))
}

func (m Model) pickerView() string {
	var b strings.Builder
	b.WriteString(accent.Render("Источник звука") + "\n\n")
	items := []string{"вся система"}
	for _, a := range m.apps {
		items = append(items, fmt.Sprintf("%s (pid %d)", a.Name, a.PID))
	}
	for i, it := range items {
		cursor := "  "
		if i == m.cursor {
			cursor = accent.Render("▸ ")
		}
		b.WriteString(cursor + it + "\n")
	}
	b.WriteString("\n" + dim.Render("[↑↓] выбрать  [enter] ок  [esc] отмена"))
	return b.String()
}
```

```go
package main

import (
	"context"
	"sync"
	"time"

	"lampa/internal/audio"
	"lampa/internal/config"
	"lampa/internal/discovery"
	"lampa/internal/engine"

// controller adds audio-source switching and discovery to the engine.
type controller struct {
	*engine.Engine
	tapPath string
	cfgPath string
	blocks  chan audio.Block
	dry     bool

	mu        sync.Mutex
	cfg       config.Config
	srcCancel context.CancelFunc
	srcName   string
	srcErr    string
}

// tapError records the helper's last error so the TUI can show it
// (stderr is invisible under the alt screen).
func (c *controller) tapError(err error) {
	c.mu.Lock()
	c.srcErr = err.Error()
	c.mu.Unlock()
}

func (c *controller) Sources(ctx context.Context) ([]audio.App, error) {
	return audio.ListApps(ctx, c.tapPath)
}

func (c *controller) SourceName() string {
	c.mu.Lock()
	defer c.mu.Unlock()
	if c.srcErr != "" && !c.View().AudioAlive {
		return c.srcName + " · " + c.srcErr
	}
	return c.srcName
}

// SetSource (re)starts the tap; nil = whole system.
func (c *controller) SetSource(app *audio.App) {
	c.mu.Lock()
	defer c.mu.Unlock()
	if c.srcCancel != nil {
		c.srcCancel()
	}
	ctx, cancel := context.WithCancel(context.Background())
	c.srcCancel = cancel
	src := &audio.TapSource{Path: c.tapPath, OnError: c.tapError}
	c.srcName, c.cfg.SourceBundle, c.srcErr = "вся система", "", ""
	if app != nil {
		src.PID = app.PID
		c.srcName, c.cfg.SourceBundle = app.Name, app.BundleID
	}
	go src.Run(ctx, c.blocks)
}

func (c *controller) Discover(ctx context.Context) (int, error) {
	found, err := discovery.Discover(ctx, 2*time.Second)
	if err != nil {
		return 0, err
	}
	c.mu.Lock()
	before := len(c.cfg.Lamps)
	added := c.cfg.Merge(found)
	newLamps := append([]config.Lamp(nil), c.cfg.Lamps[before:]...)
	err = config.Save(c.cfgPath, c.cfg)
	c.mu.Unlock()
	if !c.dry && len(newLamps) > 0 {
		for _, l := range buildLights(config.Config{Lamps: newLamps}, false) {
			c.AddLight(l)
		}
	}
	return added, err
}

func (c *controller) stop() {
	c.mu.Lock()
	defer c.mu.Unlock()
	if c.srcCancel != nil {
		c.srcCancel()
	}
}
```

```go
	"context"

	"lampa/internal/tui"

func main() {
	if len(os.Args) > 1 && os.Args[1] == "discover" {
		os.Exit(runDiscover(os.Args[2:]))
	}
	os.Exit(run(os.Args[1:]))

func info(l config.Lamp) light.Info {
	return light.Info{ID: l.ID, Name: l.Name, Model: l.Model, Role: l.Role, MaxFPS: l.MaxFPS}
}

func buildLights(cfg config.Config, dry bool) []light.Light {
	var out []light.Light
	for _, l := range cfg.Lamps {
		if l.Disabled {
			continue
		}
		if dry {
			i := info(l)
			i.Kind = light.KindVirtual
			out = append(out, light.NewVirtual(i))
			continue
		}
		switch light.Kind(l.Kind) {
		case light.KindWiZ:
			b, err := wiz.Dial(l.IP, info(l))
			if err != nil {
				fmt.Fprintf(os.Stderr, "lampa: %s: %v\n", l.Name, err)
				continue
			}
			out = append(out, b)
		case light.KindYeelight:
			out = append(out, yeelight.New(l.IP, info(l), l.Music))
		default:
			fmt.Fprintf(os.Stderr, "lampa: %s: unknown kind %q, skipped\n", l.Name, l.Kind)
		}
	}
	if len(cfg.Lamps) == 0 {
		for i := 1; i <= 4; i++ {
			out = append(out, light.NewVirtual(light.Info{ID: fmt.Sprint("v", i), Name: fmt.Sprint("virtual ", i), MaxFPS: 30}))
		}
	}
	return out
}

func findTap(flagPath string) string {
	if flagPath != "" {
		return flagPath
	}
	if exe, err := os.Executable(); err == nil {
		p := filepath.Join(filepath.Dir(exe), "lampa-tap")
		if _, err := os.Stat(p); err == nil {
			return p
		}
	}
	if p, err := exec.LookPath("lampa-tap"); err == nil {
		return p
	}
	return "lampa-tap"
}

func paletteIndex(name string) int {
	for i, p := range color.Palettes {
		if p.Name == name {
			return i
		}
	}
	return 0
}

func run(args []string) int {
	fs := flag.NewFlagSet("lampa", flag.ExitOnError)
	dry := fs.Bool("dry", false, "no real lamps: virtual lamps on screen only")
	wavPath := fs.String("wav", "", "play this WAV file instead of system audio")
	cfgPath := fs.String("config", config.DefaultPath(), "config file")
	tapFlag := fs.String("tap", "", "path to lampa-tap")
	fs.Parse(args)

	cfg, err := config.Load(*cfgPath)
	if err != nil {
		fmt.Fprintf(os.Stderr, "lampa: config: %v\n", err)
		return 1
	}
	eng := engine.New(engine.Options{
		Lights:      buildLights(cfg, *dry),
		Mode:        modes.Index(cfg.Mode),
		Palette:     paletteIndex(cfg.Palette),
		Sensitivity: cfg.Sensitivity,
		Latency:     time.Duration(cfg.LatencyMS) * time.Millisecond,
	})
	ctx, cancel := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer cancel()
	eng.Start(ctx)

	ctl := &controller{Engine: eng, tapPath: findTap(*tapFlag), cfgPath: *cfgPath,
		blocks: make(chan audio.Block, 64), cfg: cfg, dry: *dry}

	if *wavPath != "" {
		f, err := os.Open(*wavPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "lampa: %v\n", err)
			return 1
		}
		rate, samples, err := audio.ReadWAV(f)
		f.Close()
		if err != nil {
			fmt.Fprintf(os.Stderr, "lampa: %v\n", err)
			return 1
		}
		ctl.srcName = filepath.Base(*wavPath)
		go (&audio.WAVSource{Rate: rate, Samples: samples, Realtime: true}).Run(ctx, ctl.blocks)
	} else {
		var app *audio.App
		if cfg.SourceBundle != "" { // reopen the last chosen app if it is playing
			if apps, err := audio.ListApps(ctx, ctl.tapPath); err == nil {
				for i := range apps {
					if apps[i].BundleID == cfg.SourceBundle {
						app = &apps[i]
					}
				}
			}
		}
		ctl.SetSource(app)
	}
	go eng.Run(ctx, ctl.blocks)

	p := tea.NewProgram(tui.New(ctl), tea.WithAltScreen(), tea.WithContext(ctx))
	_, runErr := p.Run()

	cancel()
	ctl.stop()
	closeCtx, closeCancel := context.WithTimeout(context.Background(), 4*time.Second)
	eng.Close(closeCtx)
	closeCancel()

	v := eng.View()
	ctl.mu.Lock()
	cfg = ctl.cfg
	ctl.mu.Unlock()
	cfg.Mode, cfg.Palette = v.Mode, v.Palette
	cfg.Sensitivity, cfg.LatencyMS = v.Sensitivity, int(v.Latency.Milliseconds())
	if err := config.Save(*cfgPath, cfg); err != nil {
		fmt.Fprintf(os.Stderr, "lampa: save config: %v\n", err)
	}
	if runErr != nil && ctx.Err() == nil {
		fmt.Fprintf(os.Stderr, "lampa: %v\n", runErr)
		return 1
	}
	return 0
}

func runDiscover(args []string) int {
	fs := flag.NewFlagSet("discover", flag.ExitOnError)
	save := fs.Bool("save", false, "add found lamps to the config")
	cfgPath := fs.String("config", config.DefaultPath(), "config file")
	fs.Parse(args)
	found, err := discovery.Discover(context.Background(), 3*time.Second)
	if err != nil {
		fmt.Fprintf(os.Stderr, "lampa: %v\n", err)
		return 1
	}
	if len(found) == 0 {
		fmt.Println("ламп не найдено (WiZ: включено ли локальное управление? Yeelight: LAN Control в приложении)")
	}
	for _, f := range found {
		music := ""
		if f.Kind == light.KindYeelight {
			music = " music mode: нет"
			if f.Music {
				music = " music mode: да"
			}
		}
		fmt.Printf("%-9s %-16s %-20s %s%s\n", f.Kind, f.IP, f.Name, f.Model, music)
	}
	if *save {
		cfg, err := config.Load(*cfgPath)
		if err != nil {
			fmt.Fprintf(os.Stderr, "lampa: %v\n", err)
			return 1
		}
		n := cfg.Merge(found)
		if err := config.Save(*cfgPath, cfg); err != nil {
			fmt.Fprintf(os.Stderr, "lampa: %v\n", err)
			return 1
		}
		fmt.Printf("добавлено в %s: %d\n", *cfgPath, n)
	}
	return 0
}
```

````markdown
# lampa
````
